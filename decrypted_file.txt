<script>
    let e_file_b64 = "";
    let original_filename = "index.html";

    function openTab(evt, name) {
        document.querySelectorAll('.section').forEach(s => s.classList.remove('active'));
        document.querySelectorAll('.tab-btn').forEach(b => b.classList.remove('active'));
        document.getElementById(name).classList.add('active');
        evt.currentTarget.classList.add('active');
    }

    // Encryption File Logic
    document.getElementById('e_file').onchange = function(e) {
        let file = e.target.files[0];
        if(file) {
            original_filename = file.name;
            document.getElementById('e_file_name').innerText = "✅ " + file.name;
            let reader = new FileReader();
            reader.onload = (ev) => e_file_b64 = ev.target.result;
            reader.readAsDataURL(file);
        }
    };

    // Decryption File Logic
    document.getElementById('d_file').onchange = function(e) {
        let file = e.target.files[0];
        if(file) {
            document.getElementById('d_file_name').innerText = "✅ " + file.name;
            let reader = new FileReader();
            reader.onload = (ev) => document.getElementById('d_data').value = ev.target.result;
            reader.readAsText(file);
        }
    };

    function startEncryption() {
        const data = e_file_b64 || document.getElementById('e_data').value;
        const pass = document.getElementById('e_pass').value;
        const dom = document.getElementById('e_domain').value.trim();

        if(!data || !pass) { alert("ডাটা এবং পাসওয়ার্ড দিন!"); return; }

        const encrypted = CryptoJS.AES.encrypt(data, pass).toString();
        const obfuscatedKey = btoa(pass);

        const template = `<!DOCTYPE html><html><head><meta charset="UTF-8"><title>Locked File</title><script src="https://cdnjs.cloudflare.com/ajax/libs/crypto-js/4.1.1/crypto-js.min.js"><\/script></head><body style="background:#0f172a;color:#fff;display:flex;justify-content:center;align-items:center;height:100vh;margin:0;font-family:sans-serif;"><div style="background:#1e293b;padding:40px;border-radius:20px;text-align:center;box-shadow:0 10px 30px rgba(0,0,0,0.5); border:1px solid #334155;"><h2>🔒 Protected Content</h2><p>This file is password protected</p><input type="password" id="pw" placeholder="Enter Password" style="padding:12px;border-radius:8px;border:none;width:220px;margin-bottom:20px;text-align:center;"><br><button onclick="unlock()" style="padding:12px 30px;background:linear-gradient(to right,#4f46e5,#7c3aed);color:#fff;border:none;border-radius:8px;cursor:pointer;font-weight:bold;">Unlock Now</button></div><script>const secretData="${encrypted}";const bdom="${dom}";const bkey="${obfuscatedKey}";window.onload=()=>{if(bdom && window.location.hostname.includes(bdom))unlock(atob(bkey))};function unlock(p){const k=p||document.getElementById('pw').value;try{const dec=CryptoJS.AES.decrypt(secretData,k).toString(CryptoJS.enc.Utf8);if(dec){if(dec.startsWith('data:')){const a=document.createElement('a');a.href=dec;a.download="${original_filename}";a.click()}else{document.open();document.write(dec);document.close()}}else{alert("Wrong Password!")}}catch(e){alert("Error Decrypting!")}}<\/script></body></html>`;

        showModal("এনক্রিপশন সফল!", template, "locked_file.html");
    }

    function startDecryption() {
        let input = document.getElementById('d_data').value.trim();
        const pass = document.getElementById('d_pass').value;

        if(!input || !pass) { alert("কোড এবং পাসওয়ার্ড দিন!"); return; }

        // --- গুরুত্বপূর্ণ অংশ: কোড থেকে secretData খুঁজে বের করা ---
        let encryptedStr = "";
        if (input.includes('const secretData="')) {
            // যদি পুরো HTML পেস্ট করা হয়, তবে শুধু secretData ভ্যালুটি নিবে
            const match = input.match(/const secretData\s*=\s*["']([^"']+)["']/);
            if (match && match[1]) {
                encryptedStr = match[1];
            }
        } else {
            // যদি শুধু কোডটুকু পেস্ট করা হয়
            encryptedStr = input.replace(/^["']|["']$/g, '');
        }

        if (!encryptedStr) {
            alert("সঠিক এনক্রিপ্টেড কোড পাওয়া যায়নি!");
            return;
        }

        try {
            const bytes = CryptoJS.AES.decrypt(encryptedStr, pass);
            const decrypted = bytes.toString(CryptoJS.enc.Utf8);

            if (decrypted) {
                showModal("ডিক্রিপশন সফল!", decrypted, "decrypted_file.txt");
            } else {
                alert("ভুল পাসওয়ার্ড অথবা ডাটা সঠিক নয়!");
            }
        } catch (e) {
            alert("ডিক্রিপশন ব্যর্থ হয়েছে! কোডটি ঠিকভাবে কপি হয়েছে কি না চেক করুন।");
        }
    }

    function showModal(title, content, dlName) {
        document.getElementById('m_title').innerText = title;
        document.getElementById('m_res').value = content;
        document.getElementById('modal').style.display = 'flex';
        
        document.getElementById('m_down').onclick = () => {
            const blob = new Blob([content], {type: "text/plain"});
            const url = URL.createObjectURL(blob);
            const a = document.createElement('a');
            a.href = url;
            a.download = dlName;
            a.click();
        };
    }

    function closeModal() { document.getElementById('modal').style.display = 'none'; }
    
    function copyText() {
        const res = document.getElementById('m_res');
        res.select();
        document.execCommand('copy');
        alert("কপি করা হয়েছে!");
    }
